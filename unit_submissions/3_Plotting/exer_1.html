# ============================================================================ # Bonus Challenge (Optional) # ============================================================================ # Create a scatterplot of any two continuous variables from msleep. # Add a line of best fit using abline() and lm(). # Color points based on a condition you define (like we did with sleep_total >= 17 # in the tutorial), and add a legend to explain the colors.

Your code here:

# Condition for coloring
long_sleeper <- msleep$sleep_total >= 12
point_cols <- ifelse(long_sleeper, "red", "blue")

# Scatterplot
plot(msleep$sleep_total, msleep$sleep_rem,
     pch = 16,
     col = point_cols,
     xlab = "Total Sleep (hours)",
     ylab = "REM Sleep (hours)",
     main = "REM Sleep vs Total Sleep")

# Line of best fit
abline(lm(sleep_rem ~ sleep_total, data = msleep), lwd = 2)

# Legend
legend("topleft",
       legend = c("Total sleep >= 12 hrs", "Total sleep < 12 hrs"),
       col = c("red", "blue"),
       pch = 16,
       bty = "n")
